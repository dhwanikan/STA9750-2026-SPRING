Introduction

I grew up on Long Island,Melville, to be specific and I’ve spent most of my life watching people leave. Teachers retiring to Florida,cousins following tech jobs to Austin.Family friends priced out of Suffolk County and resettling in the Carolinas. For as long as I can remember, New York has been a state people move through rather than move to: a gateway that opens outward as much as it pulls inward.

This project gave me a data-driven reason to finally look at that pattern closely. Using American Community Survey migration flow data and a simple population projection model, I’ll examine where New Yorkers are going, who’s coming to take their place, and what our state’s congressional delegation might look like after the 2030 census. And then wearing the hat of a policy analyst advising the Governor, I’ll make the case for how New York could win back some of those seats, and the people behind them.

The analysis unfolds in four parts: data acquisition and cleaning, exploratory analysis, a population forecasting model, and a policy recommendation for an advertising strategy targeting key migration corridors. I’ve also tackled each of the four extra credit opportunities.

state_to_state_migration_2024 <- function() {

  raw <- read_excel(FILE_2024, sheet = "Table",
                    skip = 8, col_names = FALSE)
  names(raw) <- c("state_current_name", "state_1y_name", "population", "moe")

  flows_long <- raw |>
    filter(population != "X") |>
    mutate(
      state_current = name_to_abb[state_current_name],
      state_1y = case_when(
        state_1y_name %in% names(name_to_abb) ~ name_to_abb[state_1y_name],
        str_detect(tolower(state_1y_name),
                   "abroad|foreign|island|puerto|territory") ~ "Abroad",
        TRUE ~ NA_character_
      ),
      population   = if_else(is.na(population) | population == "N",
                             "0", as.character(population)) |> as.integer(),
      year_current = 2024L,
      year_1y      = 2023L
    ) |>
    filter(!is.na(state_current), !is.na(state_1y),
           state_current %in% state.abb,
           state_1y %in% c(state.abb, "Abroad")) |>
    select(state_current, state_1y, population, year_current, year_1y)

  supp_raw <- read_excel(FILE_2024,
                         sheet = "Supplemental - Current Res",
                         skip = 5, col_names = FALSE)

  supp_clean <- supp_raw |>
    select(state_current_name = 2, stayed_state = 4, stayed_county = 5) |>
    filter(state_current_name %in% state.name) |>
    mutate(
      state_current = name_to_abb[state_current_name],
      stayed_state  = if_else(is.na(stayed_state) |
                                as.character(stayed_state) == "N",
                              "0", as.character(stayed_state)),
      stayed_county = if_else(is.na(stayed_county) |
                                as.character(stayed_county) == "N",
                              "0", as.character(stayed_county)),
      population    = as.integer(stayed_state) + as.integer(stayed_county),
      state_1y      = state_current,
      year_current  = 2024L,
      year_1y       = 2023L
    ) |>
    filter(!is.na(state_current)) |>
    select(state_current, state_1y, population, year_current, year_1y)

  bind_rows(supp_clean, flows_long)
}

state_to_state_migration_2023 <- function() {

  states_with_dc <- c(state.name[1:8], "District of Columbia", state.name[9:50])

  # Part A: stayed-in-state totals
  stayed_raw <- read_excel(
    FILE_2023, range = "A8:F61",
    col_names = c("state_name", "x1", "x2", "x3", "stayed_state", "stayed_county")
  )

  stayed_clean <- stayed_raw |>
    filter(state_name %in% state.name) |>
    mutate(
      state_current = state.abb[match(state_name, state.name)],
      stayed_state  = if_else(is.na(stayed_state) |
                                as.character(stayed_state) == "N",
                              "0", as.character(stayed_state)),
      stayed_county = if_else(is.na(stayed_county) |
                                as.character(stayed_county) == "N",
                              "0", as.character(stayed_county)),
      population    = as.integer(stayed_state) + as.integer(stayed_county),
      state_1y      = state_current,
      year_current  = 2023L,
      year_1y       = 2022L
    ) |>
    select(state_current, state_1y, population, year_current, year_1y)

  # Part B: cross-state flows
  flow_col_names <- vctrs::vec_interleave(states_with_dc, NA_character_)

  flows_raw <- read_excel(FILE_2023, range = "J8:DG61",
                           col_names = flow_col_names)

  flows_long <- flows_raw |>
    select(any_of(states_with_dc)) |>
    slice(seq_len(length(states_with_dc))) |>
    mutate(state_to_name = states_with_dc) |>
    pivot_longer(cols = -state_to_name,
                 names_to = "state_from_name", values_to = "population") |>
    filter(
      state_to_name   != "District of Columbia",
      state_from_name != "District of Columbia",
      state_to_name   != state_from_name,
      !is.na(population)
    ) |>
    mutate(
      state_current = state.abb[match(state_to_name,   state.name)],
      state_1y      = state.abb[match(state_from_name, state.name)],
      population    = if_else(as.character(population) == "N", "0",
                               as.character(population)) |> as.integer(),
      year_current  = 2023L,
      year_1y       = 2022L
    ) |>
    filter(!is.na(state_current), !is.na(state_1y)) |>
    select(state_current, state_1y, population, year_current, year_1y)

  # Part C: abroad totals — read state name col + DH col together to match rows
  abroad_clean <- read_excel(FILE_2023, range = "A8:A61",
                              col_names = "state_name") |>
    bind_cols(
      read_excel(FILE_2023, range = "DH8:DH61", col_names = "abroad_total")
    ) |>
    filter(state_name %in% state.name) |>
    mutate(
      state_current = state.abb[match(state_name, state.name)],
      abroad_total  = if_else(is.na(abroad_total) |
                                as.character(abroad_total) == "N",
                              "0", as.character(abroad_total)),
      population    = as.integer(abroad_total),
      state_1y      = "Abroad",
      year_current  = 2023L,
      year_1y       = 2022L
    ) |>
    select(state_current, state_1y, population, year_current, year_1y)

  bind_rows(stayed_clean, flows_long, abroad_clean)
}

move_row_to_colnames <- function(X, row = 1) {
  X <- as_tibble(X, .name_repair = "unique")
  Xrow <- X[row, ]
  X <- X[-row, ]
  colnames(X) <- Xrow
  readr::type_convert(X)
}

pull_state_from_metro <- function(metro_name) {
  str_extract(metro_name, ".*, (\\S{2})[-[:alpha:]]* Metro Area", group = 1)
}

fit_model_params <- function(mig_data, pop_data, yr_current, yr_prior) {

  pop_cur <- pop_data |> filter(year == yr_current) |>
    mutate(s = state.abb[match(state, state.name)]) |>
    filter(!is.na(s)) |> select(state = s, pop_t1 = population)

  pop_prev <- pop_data |> filter(year == yr_prior) |>
    mutate(s = state.abb[match(state, state.name)]) |>
    filter(!is.na(s)) |> select(state = s, pop_t = population)

  M_intl <- mig_data |>
    filter(year_current == yr_current, state_1y == "Abroad") |>
    summarise(M = sum(population, na.rm = TRUE)) |> pull(M)

  g_bar <- (sum(pop_cur$pop_t1) - sum(pop_prev$pop_t) - M_intl) /
           sum(pop_prev$pop_t)

  m_ij <- mig_data |>
    filter(year_current == yr_current,
           state_1y != state_current, state_1y != "Abroad",
           state_1y %in% state.abb, state_current %in% state.abb) |>
    select(state_i = state_current, state_j = state_1y, M_ji = population) |>
    left_join(pop_prev |> rename(state_i = state, pop_i = pop_t),
              by = "state_i") |>
    left_join(pop_prev |> rename(state_j = state, pop_j = pop_t),
              by = "state_j") |>
    mutate(m_ji = M_ji / sqrt(pop_i * pop_j)) |>
    select(state_i, state_j, m_ji)

  list(g_bar = g_bar, m_ij = m_ij, pop_cur = pop_cur)
}

project_one_year <- function(pop_now, m_ij, g_bar) {
  pop_now <- pop_now |> rename(state_i = state, pop_t = pop_t1)

  mig_in <- m_ij |>
    left_join(pop_now |> select(state_i, pop_i = pop_t), by = "state_i") |>
    left_join(pop_now |> rename(state_j = state_i, pop_j = pop_t),
              by = "state_j") |>
    mutate(flow = m_ji * sqrt(pop_i * pop_j)) |>
    group_by(state_i) |>

    left_join(mig_in, by = "state_i") |>
    mutate(pop_t1 = pop_t * (1 + g_bar) + replace_na(mig_in, 0)) |>
    select(state = state_i, pop_t1)

huntington_hill <- function(pop_df, n = 435L, max_seats = 100L) {
  pop_df |>
    expand_grid(cd = seq(1L, max_seats)) |>
    mutate(
      hh_denom  = sqrt(cd * (cd - 1L)),

    mutate(pop_per_seat = pop_2030 / n_seats) |>

Extra Credit 01: The Case for a Bigger House

The US House of Representatives has had 435 members since 1929, the same year the stock market crashed and there were 123 million Americans. Today there are 335 million of us crammed into the same legislative structure. The result is a House that is systematically underrepresentative, and New York suffers most.

Consider the equity problem first. A House seat in Wyoming represents about 578,000 people, while one in New York represents nearly 780,000. That gap is not gerrymandering,it’s a structural artifact of forcing a diverse, large-population democracy into an undersized vessel. A larger House from 573 members under the Wyoming Rule to 700+ in more ambitious reforms would directly compress this disparity. New York would benefit most: more appropriations presence, more committee assignments, more leverage over transit funding, healthcare, and financial regulation.

Second, consider democratic culture. When each House seat is scarce, the incentive to protect it through gerrymandering is overwhelming. Smaller districts mean more coherent communities and more accountable representatives. For a heterogeneous state like New York: rural Adirondack counties alongside dense Brooklyn precincts, this matters enormously. The argument is not partisan: scholars on both the left and right have made the case for expansion. What we’re proposing is simply to stop treating 435 as sacred. The Constitution sets no upper limit. It’s a statute, not scripture, last updated when Herbert Hoover was president.

project_stateg <- function(pop_now, m_ij, g_rates, g_nat) {
  pop_now <- pop_now |> rename(state_i = state, pop_t = pop_t1)

  mig_in <- m_ij |>
    left_join(pop_now |> select(state_i, pop_i = pop_t), by = "state_i") |>
    left_join(pop_now |> rename(state_j = state_i, pop_j = pop_t),
              by = "state_j") |>
    mutate(flow = m_ji * sqrt(pop_i * pop_j)) |>
    group_by(state_i) |>
    summarise(mig_in = sum(flow, na.rm = TRUE), .groups = "drop")

  pop_now |>
    left_join(g_rates |> select(state, g_i),
              by = c("state_i" = "state")) |>
    left_join(mig_in, by = "state_i") |>
    mutate(
      g_use  = replace_na(g_i, g_nat),
      pop_t1 = pop_t * (1 + g_use) + replace_na(mig_in, 0)
    ) |>
    select(state = state_i, pop_t1)
}

### Task 7: Fitting the Gravity Model

$$P_i^{(t+1)} = P_i^{(t)}(1 + \bar{g}) + \sum_j m_{j \to i} \sqrt{P_i^{(t)} P_j^{(t)}}$$

```{r}
#| label: task7-params

fit_model_params <- function(mig_data, pop_data, yr_current, yr_prior) {

  pop_cur <- pop_data |> filter(year == yr_current) |>
    mutate(s = state.abb[match(state, state.name)]) |>
    filter(!is.na(s)) |> select(state = s, pop_t1 = population)

  pop_prev <- pop_data |> filter(year == yr_prior) |>
    mutate(s = state.abb[match(state, state.name)]) |>
    filter(!is.na(s)) |> select(state = s, pop_t = population)

  M_intl <- mig_data |>
    filter(year_current == yr_current, state_1y == "Abroad") |>
    summarise(M = sum(population, na.rm = TRUE)) |> pull(M)

  g_bar <- (sum(pop_cur$pop_t1) - sum(pop_prev$pop_t) - M_intl) /
           sum(pop_prev$pop_t)

  m_ij <- mig_data |>
    filter(year_current == yr_current,
           state_1y != state_current, state_1y != "Abroad",
           state_1y %in% state.abb, state_current %in% state.abb) |>
    select(state_i = state_current, state_j = state_1y, M_ji = population) |>
    left_join(pop_prev |> rename(state_i = state, pop_i = pop_t),
              by = "state_i") |>
    left_join(pop_prev |> rename(state_j = state, pop_j = pop_t),
              by = "state_j") |>
    mutate(m_ji = M_ji / sqrt(pop_i * pop_j)) |>
    select(state_i, state_j, m_ji)

  list(g_bar = g_bar, m_ij = m_ij, pop_cur = pop_cur)
}
```

```{r}
huntington_hill <- function(pop_df, n = 435L, max_seats = 100L) {
  pop_df |>
    expand_grid(cd = seq(1L, max_seats)) |>
    mutate(
      hh_denom  = sqrt(cd * (cd - 1L)),
      hh_denom  = if_else(cd == 1L, 1, hh_denom),
      hh_weight = pop_t1 / hh_denom
    ) |>
    slice_max(hh_weight, n = n) |>
    group_by(state) |>
    summarise(pop_2030 = first(pop_t1), n_seats = n(), .groups = "drop") |>
    mutate(pop_per_seat = pop_2030 / n_seats) |>
    arrange(desc(n_seats), desc(pop_2030))
}
```

## Extra Credit 01: The Case for a Bigger House

The US House of Representatives has had 435 members since 1929, the same year the stock market crashed and there were 123 million Americans. Today there are 335 million of us crammed into the same legislative structure. The result is a House that is systematically underrepresentative, and New York suffers most.

Consider the equity problem first. A House seat in Wyoming represents about 578,000 people, while one in New York represents nearly 780,000. That gap is not gerrymandering,it's a structural artifact of forcing a diverse, large-population democracy into an undersized vessel. A larger House from 573 members under the Wyoming Rule to 700+ in more ambitious reforms would directly compress this disparity. New York would benefit most: more appropriations presence, more committee assignments, more leverage over transit funding, healthcare, and financial regulation.

Second, consider democratic culture. When each House seat is scarce, the incentive to protect it through gerrymandering is overwhelming. Smaller districts mean more coherent communities and more accountable representatives. For a heterogeneous state like New York: rural Adirondack counties alongside dense Brooklyn precincts, this matters enormously. The argument is not partisan: scholars on both the left and right have made the case for expansion. What we're proposing is simply to stop treating 435 as sacred. The Constitution sets no upper limit. It's a statute, not scripture, last updated when Herbert Hoover was president.

```{r}
#| label: ec03-project

project_stateg <- function(pop_now, m_ij, g_rates, g_nat) {
  pop_now <- pop_now |> rename(state_i = state, pop_t = pop_t1)

  mig_in <- m_ij |>
    left_join(pop_now |> select(state_i, pop_i = pop_t), by = "state_i") |>
    left_join(pop_now |> rename(state_j = state_i, pop_j = pop_t),
              by = "state_j") |>
    mutate(flow = m_ji * sqrt(pop_i * pop_j)) |>
    group_by(state_i) |>
    summarise(mig_in = sum(flow, na.rm = TRUE), .groups = "drop")

  pop_now |>
    left_join(g_rates |> select(state, g_i),
              by = c("state_i" = "state")) |>
    left_join(mig_in, by = "state_i") |>
    mutate(
      g_use  = replace_na(g_i, g_nat),
      pop_t1 = pop_t * (1 + g_use) + replace_na(mig_in, 0)
    ) |>
    select(state = state_i, pop_t1)
}
```
